#!/usr/bin/env bash
# scope=perf: record which BUILD_ID a probe hit returns (httpload.mjs regex missed it). Single light request, port 3342.
W=/mnt/ssd/afldb-rem-perf-20261114
export PATH="$HOME/.nvm/versions/node/v22.23.2/bin:$PATH"
export AFLDB_V2_PG_ROOT="$W/pg-root" AFLDB_V2_PG_BIN=/usr/lib/postgresql/16/bin AFLDB_V2_PG_PORT=55621
PORT=3342
ss -ltn | awk '{print $4}' | grep -E ":$PORT$" && { echo busy; exit 1; }
cd $W/cand
echo "BUILD_ID file: $(cat .next/BUILD_ID)  standalone: $(cat .next/standalone/.next/BUILD_ID 2>/dev/null)"
(exec node --import tsx tools/v2/isolated-pg.ts run afldb_v2_remperf_a_test -- env PORT=$PORT HOSTNAME=127.0.0.1 NODE_ENV=production node .next/standalone/server.js > $W/evidence/server-buildid.log 2>&1) &
for i in $(seq 1 60); do curl -s -o /dev/null http://127.0.0.1:$PORT/ && break; sleep 1; done
curl -s http://127.0.0.1:$PORT/clubs -o $W/evidence/clubs.html -w "status=%{http_code}\n"
echo "ids in html:"; grep -o '"b\\":\\"[A-Za-z0-9_-]*' $W/evidence/clubs.html | sort -u | head; grep -o '_next/static/[A-Za-z0-9_-]\{15,30\}/_[a-zA-Z]*Manifest' $W/evidence/clubs.html | sort -u | head
echo "contains BUILD_ID: $(grep -c "$(cat .next/BUILD_ID)" $W/evidence/clubs.html)"
LP=$(ss -ltnp 2>/dev/null | grep ":$PORT " | grep -o "pid=[0-9]*" | head -1 | cut -d= -f2); [ -n "$LP" ] && kill -TERM $LP; sleep 2
ss -ltn | awk '{print $4}' | grep -E ":$PORT$" && echo "still listening" || echo "port $PORT closed"
rm -f $W/evidence/clubs.html
