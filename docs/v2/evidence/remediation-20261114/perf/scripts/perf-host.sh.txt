#!/usr/bin/env bash
# perf agent host setup (remediation run 2026-11-14). Phases: setup | cluster | template
set -u -o pipefail
W=/mnt/ssd/afldb-rem-perf-20261114
export PATH="$HOME/.nvm/versions/node/v22.23.2/bin:$PATH"
DUMP=/mnt/ssd/afldb/workspaces/afldb-issue333-o2-20261104/private/afldb_prod-20261104T021037Z.dump
DUMP_SHA=e77c307d3e0cddcb652e398a946e4c22c2063f13c70edadf96a975c92842826d
DUMP_BYTES=30141895
PORT=55621
TMPL=afldb_v2_remperf_tmpl_test
CAND="$W/cand"; EV="$W/evidence"
export AFLDB_V2_PG_ROOT="$W/pg-root" AFLDB_V2_PG_BIN=/usr/lib/postgresql/16/bin AFLDB_V2_PG_PORT=$PORT
export PYTHON=/home/arm/i250/afldb/.venv/bin/python AFLDB_PYTHON=/home/arm/i250/afldb/.venv/bin/python
IR() { (cd "$CAND" && node --import tsx tools/v2/isolated-pg.ts "$@"); }
RUNDB() { local db="$1"; shift; (cd "$CAND" && node --import tsx tools/v2/isolated-pg.ts run "$db" -- "$@"); }
strip() { sed 's/\x1b\[[0-9;]*m//g' | grep -v -i 'postgres://'; }
listener_is_ours() {
  local pmpid; pmpid=$(head -1 "$AFLDB_V2_PG_ROOT/data/postmaster.pid")
  ss -ltnp 2>/dev/null | grep ":$PORT " > "$EV/listener.txt"
  cat "$EV/listener.txt"; echo "postmaster pid $pmpid"
  [ "$(grep -c "127.0.0.1:$PORT " "$EV/listener.txt")" = 1 ] || { echo "not exactly one listener"; return 1; }
  [ "$(grep -c "pid=$pmpid," "$EV/listener.txt")" = 1 ] || { echo "listener not ours"; return 1; }
}
case "${1:-}" in
setup)
  mkdir -p "$W/in" "$EV"
  cp /mnt/ssd/afldb-review-20261114T0640Z/in/restore.mjs /mnt/ssd/afldb-review-20261114T0640Z/in/probe-dbs.mjs "$W/in/"
  ss -ltn | awk '{print $4}' | grep -E ":$PORT$" && { echo "port in use"; exit 1; }
  [ "$(stat -c %s "$DUMP")" = "$DUMP_BYTES" ] || { echo size mismatch; exit 1; }
  [ "$(sha256sum "$DUMP" | cut -d' ' -f1)" = "$DUMP_SHA" ] || { echo sha mismatch; exit 1; }
  echo "snapshot verified"
  git clone -q /mnt/ssd/afldb-review-20261114T0640Z/cand "$CAND" || exit 1
  cd "$CAND" && git checkout -q --detach de640826d13489c58392563e7ae1ff62403ed293 && npm ci --no-audit --no-fund 2>&1 | tail -3
  git status --porcelain | head; git rev-parse HEAD ;;
cluster)
  IR init || exit 1
  IR start > "$EV/pg-start.out" 2>&1 < /dev/null
  sleep 3
  IR roles && IR status || exit 1
  listener_is_ours ;;
template)
  IR create-db "$TMPL" || exit 1
  RUNDB "$TMPL" node "$W/in/restore.mjs" "$DUMP" "$DUMP_SHA" || exit 1
  RUNDB "$TMPL" node --import tsx tools/db/migrate.ts --target test 2>&1 | strip | tail -6
  cp "$W/in/probe-dbs.mjs" "$CAND/.probe-dbs.mjs"
  RUNDB "$TMPL" node .probe-dbs.mjs "$TMPL" | tee "$EV/tmpl-probe.txt"
  rm -f "$CAND/.probe-dbs.mjs"
  grep -q '"ledger":"118_fixtures_unconfirmed_participants.sql"' "$EV/tmpl-probe.txt" || { echo "ledger not 118"; exit 1; } ;;
*) echo usage ;;
esac
