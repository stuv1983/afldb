# remediation run 2 integration (final qualification): workspace setup (clone base, npm ci, cluster, template)
set -u -o pipefail
export W=/mnt/ssd/afldb-rem2-int-20261114
export PATH="$HOME/.nvm/versions/node/v22.23.2/bin:$PATH"
export PYTHON=/home/arm/i250/afldb/.venv/bin/python AFLDB_PYTHON=/home/arm/i250/afldb/.venv/bin/python
export AFLDB_V2_PG_ROOT=$W/pg-root AFLDB_V2_PG_BIN=/usr/lib/postgresql/16/bin AFLDB_V2_PG_PORT=55660
DUMP=/mnt/ssd/afldb/workspaces/afldb-issue333-o2-20261104/private/afldb_prod-20261104T021037Z.dump
DUMP_SHA=e77c307d3e0cddcb652e398a946e4c22c2063f13c70edadf96a975c92842826d
DUMP_BYTES=30141895
BUNDLE=/mnt/ssd/afldb-rem2-base/base-e8666c17.bundle
BUNDLE_SHA=02940af6d3065abae2e0cd42f8ca6384462a42350a293401b368d0db2453d44a
PORT=55660
CAND=$W/cand
TMPL=afldb_v2_rem2int_tmpl_test
mkdir -p $W/evidence $W/in
IR() { (cd "$CAND" && node --import tsx tools/v2/isolated-pg.ts "$@"); }
RUNDB() { local db="$1"; shift; (cd "$CAND" && node --import tsx tools/v2/isolated-pg.ts run "$db" -- "$@"); }
strip() { sed 's/\x1b\[[0-9;]*m//g' | grep -v -i 'postgres://'; }
phase_clone() {
  [ "$(stat -c %s "$DUMP")" = "$DUMP_BYTES" ] || { echo "snapshot size mismatch"; return 1; }
  [ "$(sha256sum "$DUMP" | cut -d' ' -f1)" = "$DUMP_SHA" ] || { echo "snapshot sha256 mismatch"; return 1; }
  echo "snapshot verified"
  [ "$(sha256sum "$BUNDLE" | cut -d' ' -f1)" = "$BUNDLE_SHA" ] || { echo "bundle sha256 mismatch"; return 1; }
  echo "bundle verified"
  if ss -ltn 2>/dev/null | awk '{print $4}' | grep -E ":$PORT$"; then echo "port in use"; return 1; fi
  [ -d "$CAND" ] || git clone -q /mnt/ssd/afldb-review-20261114T0640Z/cand "$CAND" || return 1
  cd "$CAND" && git fetch -q "$BUNDLE" claude/remediation-20261114:refs/heads/base && git checkout -q base && git rev-parse HEAD
  timeout 900 npm ci --no-audit --no-fund 2>&1 | tail -4
  git status --porcelain | wc -l
}
phase_cluster() {
  IR init || return 1
  IR start > "$W/evidence/pg-start.out" 2>&1 < /dev/null
  sleep 3
  IR roles && IR status || return 1
  ss -ltnp 2>/dev/null | grep ":$PORT "
}
phase_template() {
  IR create-db "$TMPL" || return 1
  flock /mnt/ssd/afldb-rem2-base/heavy.lock nice -n 10 bash -c "cd $CAND && node --import tsx tools/v2/isolated-pg.ts run $TMPL -- node /mnt/ssd/afldb-review-20261114T0640Z/in/restore.mjs $DUMP $DUMP_SHA" || return 1
  RUNDB "$TMPL" node --import tsx tools/db/migrate.ts --target test 2>&1 | strip | tail -6
  cp /mnt/ssd/afldb-review-20261114T0640Z/in/probe-dbs.mjs "$CAND/.probe-dbs.mjs"
  RUNDB "$TMPL" node .probe-dbs.mjs "$TMPL" | tee "$W/evidence/tmpl-probe.txt"
  rm -f "$CAND/.probe-dbs.mjs"
  grep -q '"ledger":"118_fixtures_unconfirmed_participants.sql"' "$W/evidence/tmpl-probe.txt" || { echo "ledger is not 118"; return 1; }
}
for ph in "$@"; do echo "== $ph"; $ph || { echo "FAILED $ph"; exit 1; }; done
echo SETUP-DONE
