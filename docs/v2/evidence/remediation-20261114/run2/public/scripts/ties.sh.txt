#!/usr/bin/env bash
# usage: ties.sh <dbsuffix> <outfile>  (measurement probe; copies the probe into cand/tests/integration, removes it after)
set -u -o pipefail
export W=/mnt/ssd/afldb-rem2-public-20261114
export PATH="$HOME/.nvm/versions/node/v22.23.2/bin:$PATH"
export AFLDB_V2_PG_ROOT=$W/pg-root AFLDB_V2_PG_BIN=/usr/lib/postgresql/16/bin AFLDB_V2_PG_PORT=55654
SUF=$1; OUT=$2
cd $W/cand
cp $W/in/ties-probe.test.ts tests/integration/zz-rem2-public-ties.test.ts
DB=afldb_v2_rem2public_${SUF}_test
node --import tsx tools/v2/isolated-pg.ts create-db "$DB" --template afldb_v2_rem2public_tmpl_test > "$OUT.mk" 2>&1 || { cat "$OUT.mk"; rm -f tests/integration/zz-rem2-public-ties.test.ts; exit 1; }
timeout 900 node --import tsx tools/v2/isolated-pg.ts run "$DB" -- node node_modules/vitest/vitest.mjs run tests/integration/zz-rem2-public-ties.test.ts > "$OUT.log" 2>&1
echo "rc=$?" >> "$OUT.log"
rm -f tests/integration/zz-rem2-public-ties.test.ts
sed 's/\x1b\[[0-9;]*m//g' "$OUT.log" | grep -a -o 'TIEPROBE.*' > "$OUT"
grep -c . "$OUT"; tail -3 "$OUT.log"
