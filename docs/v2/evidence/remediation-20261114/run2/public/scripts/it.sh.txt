#!/usr/bin/env bash
# usage: it.sh <dbsuffix> <outfile> <integration test file...>
# fresh template copy each time (dropped first if present is NOT done: pass a new suffix), isolated-pg run, bounded
set -u -o pipefail
export W=/mnt/ssd/afldb-rem2-public-20261114
export PATH="$HOME/.nvm/versions/node/v22.23.2/bin:$PATH"
export PYTHON=/home/arm/i250/afldb/.venv/bin/python AFLDB_PYTHON=/home/arm/i250/afldb/.venv/bin/python
export AFLDB_V2_PG_ROOT=$W/pg-root AFLDB_V2_PG_BIN=/usr/lib/postgresql/16/bin AFLDB_V2_PG_PORT=55654
SUF=$1; OUT=$2; shift 2
CAND=${CAND:-$W/cand}
DB=afldb_v2_rem2public_${SUF}_test
cd "$CAND"
node --import tsx tools/v2/isolated-pg.ts create-db "$DB" --template afldb_v2_rem2public_tmpl_test > "$OUT.mk" 2>&1 || { cat "$OUT.mk"; exit 1; }
timeout 1500 nice -n 10 node --import tsx tools/v2/isolated-pg.ts run "$DB" -- node node_modules/vitest/vitest.mjs run "$@" > "$OUT" 2>&1
echo "rc=$?" >> "$OUT"
sed 's/\x1b\[[0-9;]*m//g' "$OUT" | grep -a -E "^ *(FAIL|×|✗)|Test Files|Tests |AssertionError|rc=|Error:" | head -${HEADN:-40}
