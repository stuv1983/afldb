#!/usr/bin/env bash
# usage: qual3.sh <phase...>   phases: static unit integ corpus   (clean checkout $W/qual at the MED-1 code commit; results under $W/evidence/qual3)
set -u -o pipefail
export W=/mnt/ssd/afldb-rem2-public-20261114
export PATH="$HOME/.nvm/versions/node/v22.23.2/bin:$PATH"
export PYTHON=/home/arm/i250/afldb/.venv/bin/python AFLDB_PYTHON=/home/arm/i250/afldb/.venv/bin/python
export AFLDB_V2_PG_ROOT=$W/pg-root AFLDB_V2_PG_BIN=/usr/lib/postgresql/16/bin AFLDB_V2_PG_PORT=55654
Q=$W/qual; E=$W/evidence/qual3; mkdir -p $E
strip() { sed 's/\x1b\[[0-9;]*m//g' | grep -v -i 'postgres://'; }

phase_static() {
  bash $W/in/tsc-lint.sh $Q $E/static
}

phase_unit() {
  cd $Q
  timeout 1200 nice -n 10 node node_modules/vitest/vitest.mjs run \
    tests/nl-parser.test.ts tests/nl-plan.test.ts tests/nl-describe.test.ts tests/query-intent.test.ts tests/nl-audit-acceptance.test.ts \
    tests/home-records.test.ts tests/club-comparison.test.ts tests/club-comparison-view.test.ts \
    tests/unit/qualifying-matches-gate.test.ts \
    > $E/unit-focused.log 2>&1
  echo "unit-focused rc=$?" >> $E/unit-focused.log
  strip < $E/unit-focused.log | grep -a -E "Test Files|Tests |FAIL|rc=" | head -30
}

integ_one() { # <suffix> <file>
  local suf=$1 f=$2 db=afldb_v2_rem2public_$1_test
  cd $Q
  node --import tsx tools/v2/isolated-pg.ts create-db "$db" --template afldb_v2_rem2public_tmpl_test > $E/integ-$suf.mk 2>&1 || { echo "mk failed $suf"; return 1; }
  timeout 1500 nice -n 10 node --import tsx tools/v2/isolated-pg.ts run "$db" -- node node_modules/vitest/vitest.mjs run "$f" > $E/integ-$suf.log 2>&1
  echo "rc=$?" >> $E/integ-$suf.log
  echo "== $f"; strip < $E/integ-$suf.log | grep -a -E "Test Files|Tests |^ *FAIL|rc=" | head -12
}

phase_integ() {
  integ_one k1 tests/integration/nl-semantic-mapping.test.ts
  integ_one k2 tests/integration/club-comparison-route.test.ts
  integ_one k4 tests/integration/nl-answers.test.ts
  integ_one k5 tests/integration/nl-answers-game-season.test.ts
  integ_one k6 tests/integration/nl-answers-team-club.test.ts
  integ_one k7 tests/integration/nl-answer-boundary.test.ts
  integ_one k8 tests/integration/nl-vocab.test.ts
}

phase_corpus() {
  bash $W/in/corpus-parse.sh fix3 $Q corpusfix3
  cd $W/evidence
  sha256sum corpus-fix3/parse/results.jsonl | tee -a corpus-hashes3.txt
  sha256sum corpus-base/parse/results.jsonl corpus-fix/parse/results.jsonl | tee -a corpus-hashes3.txt
  echo "### fix3 (a65ca3b4) vs base (e8666c17, v68)"
  SHOW=40 node $W/in/corpus-diff.mjs corpus-base/parse/results.jsonl corpus-fix3/parse/results.jsonl corpus-diff3-vs-base.jsonl | tee corpus-diff3-vs-base.txt
  echo "### fix3 (a65ca3b4) vs fix (cb75e3b3 v69)"
  SHOW=40 node $W/in/corpus-diff.mjs corpus-fix/parse/results.jsonl corpus-fix3/parse/results.jsonl corpus-diff3-vs-cb75.jsonl | tee corpus-diff3-vs-cb75.txt
}

for ph in "$@"; do echo "##### $ph $(date -u +%T)"; phase_$ph; done
echo "##### done $(date -u +%T)"
