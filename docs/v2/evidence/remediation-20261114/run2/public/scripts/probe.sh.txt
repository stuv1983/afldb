#!/usr/bin/env bash
# usage: probe.sh <questions-file> <out-jsonl> [checkout dir]  (scope public; own cluster/port/db copy)
set -u -o pipefail
export W=/mnt/ssd/afldb-rem2-public-20261114
export PATH="$HOME/.nvm/versions/node/v22.23.2/bin:$PATH"
export AFLDB_V2_PG_ROOT=$W/pg-root AFLDB_V2_PG_BIN=/usr/lib/postgresql/16/bin AFLDB_V2_PG_PORT=55654
CAND=${3:-$W/cand}
DB=afldb_v2_rem2public_work_test
QF=$1; OUT=$2
cp /mnt/ssd/afldb-rem2-public-20261114/in/probe.test.ts "$CAND/tests/integration/zz-rem2-public-probe.test.ts"
export RV_QUESTIONS="$(paste -sd'|' "$QF")"
cd "$CAND"
timeout 600 node --import tsx tools/v2/isolated-pg.ts run $DB -- node node_modules/vitest/vitest.mjs run tests/integration/zz-rem2-public-probe.test.ts > "$OUT.log" 2>&1
echo "rc=$?" >> "$OUT.log"
sed 's/\x1b\[[0-9;]*m//g' "$OUT.log" | grep -a -o '{"q".*' > "$OUT"
rm -f "$CAND/tests/integration/zz-rem2-public-probe.test.ts"
grep -c . "$OUT"; tail -3 "$OUT.log"
