#!/usr/bin/env bash
# usage: mkdb.sh <suffix>   -> fresh afldb_v2_rem2public_<suffix>_test from the template (drops an existing one of that name first)
set -u -o pipefail
export W=/mnt/ssd/afldb-rem2-public-20261114
export PATH="$HOME/.nvm/versions/node/v22.23.2/bin:$PATH"
export AFLDB_V2_PG_ROOT=$W/pg-root AFLDB_V2_PG_BIN=/usr/lib/postgresql/16/bin AFLDB_V2_PG_PORT=55654
cd $W/cand
DB=afldb_v2_rem2public_$1_test
node --import tsx tools/v2/isolated-pg.ts create-db "$DB" --template afldb_v2_rem2public_tmpl_test
