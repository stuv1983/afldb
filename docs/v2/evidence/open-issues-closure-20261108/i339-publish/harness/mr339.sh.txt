#!/usr/bin/env bash
# ISSUE-339 merge readiness in a disposable clone laid out like the operator repo: primary checkout on a
# clean local main == recorded origin/main, linked worktree on the issue branch. Read-only checks only.
set -uo pipefail
W="$1"; EXPECT="$2"; BASE="$3"; BUNDLE="$W/afldb-i339-cand.bundle"; ROOTM="$W/mr"; EV="$W/evMR"
P=0; F=0; pass(){ echo "PASS $1"; P=$((P+1)); }; fail(){ echo "FAIL $1"; F=$((F+1)); }
check_eq(){ if [ "$2" = "$3" ]; then pass "$1"; else fail "$1 (want '$2' got '$3')"; fi; }
[ -e "$ROOTM" ] && { echo "refusing: $ROOTM exists"; exit 2; }
mkdir -p "$ROOTM" "$EV"
export NVM_DIR="$HOME/.nvm"; . "$NVM_DIR/nvm.sh" >/dev/null; nvm use --silent 22.23.2 >/dev/null
git clone -q -b claude/issue-339-gate-publish "$BUNDLE" "$ROOTM/main" || exit 2
cd "$ROOTM/main"
git fetch -q "$BUNDLE" '+refs/remotes/origin/main:refs/remotes/origin/main' || exit 2
git checkout -q -B main refs/remotes/origin/main || exit 2
git worktree add -q "$ROOTM/wt" claude/issue-339-gate-publish || exit 2
check_eq "primary main == recorded origin/main == base" "$BASE $BASE" "$(git rev-parse main) $(git rev-parse refs/remotes/origin/main)"
check_eq "issue worktree HEAD" "$EXPECT" "$(git -C "$ROOTM/wt" rev-parse HEAD)"
check_eq "branch grandparent is the base (fast-forward possible)" "$BASE" "$(git rev-parse "$EXPECT^^")"
npm ci --no-audit --no-fund > "$EV/npm-ci.log" 2>&1 && pass "npm ci" || { fail "npm ci"; exit 1; }
TSX="$ROOTM/main/node_modules/.bin/tsx"
echo "== merge:ready --issue 339 (from the issue worktree) =="
(cd "$ROOTM/wt" && "$TSX" tools/dev/merge-readiness.ts --issue 339) > "$EV/merge-ready-339.txt" 2>&1; rc=$?
grep -E '^(PASS|FAIL|WARN|INFO)|Merge readiness' "$EV/merge-ready-339.txt" | sed 's/^/  /'
check_eq "merge:ready 339 exit" 0 "$rc"
grep -q 'Merge readiness: READY' "$EV/merge-ready-339.txt" && pass "merge:ready 339 READY" || fail "merge:ready 339 not READY"
echo "== preflight --mode merge (clean primary main) =="
(cd "$ROOTM/main" && "$TSX" tools/dev/preflight.ts --mode merge) > "$EV/preflight-merge.txt" 2>&1; rc=$?
grep -E '^(PASS|FAIL|WARN|INFO)|Preflight result' "$EV/preflight-merge.txt" | sed 's/^/  /'
check_eq "preflight --mode merge exit" 0 "$rc"
echo "== fast-forward rehearsal in the clone =="
git merge -q --ff-only claude/issue-339-gate-publish && check_eq "ff-only merge result is the tested SHA" "$EXPECT" "$(git rev-parse main)" || fail "ff-only merge"
check_eq "merged tree equals tested tree" "$(git rev-parse "$EXPECT^{tree}")" "$(git rev-parse 'main^{tree}')"
echo; echo "ISSUE-339 merge readiness: $P passed, $F failed"; echo "$P/$F" > "$EV/result.txt"
