#!/usr/bin/env bash
# Initialise and start a fresh loopback-only isolated cluster for this closure run.
set -euo pipefail
source "$(dirname "$0")/env.sh"
export AFLDB_V2_PG_ROOT="${WS}-pg" AFLDB_V2_PG_PORT=55481
if [ -e "$AFLDB_V2_PG_ROOT" ]; then echo "cluster root exists: $AFLDB_V2_PG_ROOT"; fi
I="node_modules/.bin/tsx tools/v2/isolated-pg.ts"
$I init
$I start
$I roles
$I list
